Race check: GET /api/bots/:id/file while a second process swaps a desk entry as fast as it can.
Each swap is atomic (make link or file at .<name>.tmp, then rename over <name>).
3000 requests per mode, 10 at a time. "outside" means the body was /etc/os-release.

link mode: link.md flips between -> plan.md (inside) and -> /etc/os-release (outside)
file mode: target.md flips between a regular file and a link to /etc/os-release

Run: bun race.ts <link|file> http://127.0.0.1:<port> <botId> <desk dir> [requests]

    import { readFileSync, renameSync, rmSync, symlinkSync, writeFileSync } from "node:fs";
    import { join } from "node:path";
    
    const [mode, base, botId, desk, count = "3000"] = process.argv.slice(2);
    const OUTSIDE = "/etc/os-release";
    
    if (mode === "flip-link" || mode === "flip-file") {
      const name = mode === "flip-link" ? "link.md" : "target.md";
      const tmp = join(desk!, `.${name}.tmp`);
      const until = Date.now() + 60_000;
      let n = 0;
      while (Date.now() < until) {
        rmSync(tmp, { force: true });
        if (mode === "flip-link") symlinkSync(n % 2 ? OUTSIDE : "plan.md", tmp);
        else if (n % 2) symlinkSync(OUTSIDE, tmp);
        else writeFileSync(tmp, "INSIDE target.md\n");
        renameSync(tmp, join(desk!, name));
        n++;
      }
      process.exit(0);
    }
    
    const name = mode === "link" ? "link.md" : "target.md";
    writeFileSync(join(desk!, "plan.md"), "INSIDE plan.md\n");
    const flipper = Bun.spawn(["bun", import.meta.path, `flip-${mode}`, base!, botId!, desk!], { stdout: "inherit", stderr: "inherit" });
    await Bun.sleep(300);
    
    const secret = readFileSync(OUTSIDE, "utf8");
    const tally: Record<string, number> = { inside: 0, outside: 0, refused: 0, other: 0 };
    const url = `${base}/api/bots/${botId}/file?path=${encodeURIComponent(name)}`;
    for (let i = 0; i < Number(count); i += 10) {
      await Promise.all(
        Array.from({ length: 10 }, async () => {
          const res = await fetch(url);
          const text = await res.text();
          if (res.status === 404) tally.refused!++;
          else if (text === secret) tally.outside!++;
          else if (text.startsWith("INSIDE")) tally.inside!++;
          else tally.other!++;
        }),
      );
    }
    flipper.kill();
    await flipper.exited;
    console.log(`mode=${mode} path=${name} requests=${count}`, JSON.stringify(tally));
